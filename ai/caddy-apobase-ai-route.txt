42berlinaiclub.de {
	encode gzip zstd

	# Block direct static access to /data/* (SQLite DB, materials) - served only via authed API
	handle /data/* {
		respond 404
	}
	handle /ai-club/data/* {
		respond 404
	}

	# API -> Flask backend (headers set by app)
	handle /api/* {
		uri strip_prefix /api
		reverse_proxy ai-club-42berlin:5000
	}

	# Long-cache for fingerprinted static assets
	@versionedAssets path *.css *.js *.woff *.woff2 *.png *.jpg *.jpeg *.gif *.svg *.ico *.webp
	header @versionedAssets Cache-Control "public, max-age=31536000, immutable"

	# Legacy /ai-club/ subpath -> canonical root (301, keeps old links working)
	@legacyAiClub path_regexp ^/ai-club/(.*)$
	redir @legacyAiClub /{re.1} permanent

	# Apobase — pharmacy info terminal (subpath)
	# Legacy /sowedoo/ subpath -> /apobase/ (301, keeps old links working)
	@legacySowedoo path_regexp ^/sowedoo/(.*)$
	redir @legacySowedoo /apobase/{re.1} permanent

	# Apobase AI assistant API (RAG chatbot, apobase-ai container)
	handle /apobase/ai/api/* {
		uri strip_prefix /apobase/ai/api
		reverse_proxy apobase-ai:8100
		header {
			Access-Control-Allow-Origin *
			Access-Control-Allow-Methods "GET, POST, OPTIONS"
			Access-Control-Allow-Headers "Content-Type"
		}
	}

	handle /apobase/* {
		root * /srv/apobase
		uri strip_prefix /apobase
		try_files {path} {path}/index.html index.html
		file_server
		header {
			X-Content-Type-Options nosniff
			X-Frame-Options DENY

			Strict-Transport-Security "max-age=31536000; includeSubDomains"
			Permissions-Policy "camera=(), microphone=(), geolocation=(), payment=(), usb=()"
			-Server
		}
	}

	# Static frontend + security headers (root path kept for backward compatibility)
	handle {
		root * /opt/42berlinaiclub
		try_files {path} {path}/index.html
		file_server
		header {
			X-Content-Type-Options nosniff
			Referrer-Policy strict-origin-when-cross-origin
			Strict-Transport-Security "max-age=31536000; includeSubDomains"
			Permissions-Policy "camera=(), microphone=(), geolocation=(), payment=(), usb=()"

			-Server
		}
	}
	handle_errors {
		rewrite * /404.html
		file_server

			X-Content-Type-Options nosniff
